}

/* target band geometry: map angle range to the vertical strip */
let bandLo,bandHi,mapLo,mapHi;
function setupTargetBand(){
  const t=current.trackType;
  if(t==='flexion'){ mapLo=0;mapHi=150; bandLo=current.goal; bandHi=150; }     // want to reach high bend
  else if(t==='extension'){ mapLo=120;mapHi=180; bandLo=current.goal; bandHi=180; }
  else if(t==='slr'){ mapLo=0;mapHi=70; bandLo=current.goal; bandHi=70; }
  const band=$('band');
  const top=100-((bandHi-mapLo)/(mapHi-mapLo))*100;
  const bot=100-((bandLo-mapLo)/(mapHi-mapLo))*100;
  band.style.top=top+'%'; band.style.bottom=(100-bot)+'%';
}
function updateNeedle(val){
  let p=((val-mapLo)/(mapHi-mapLo)); p=Math.max(0,Math.min(1,p));
  $('needle').style.bottom=(p*100)+'%';
}

/* ---------------- Geometry ---------------- */
function angle(a,b,c){ // angle at b in degrees
  const ab={x:a.x-b.x,y:a.y-b.y}, cb={x:c.x-b.x,y:c.y-b.y};
  const dot=ab.x*cb.x+ab.y*cb.y;
  const m=Math.hypot(ab.x,ab.y)*Math.hypot(cb.x,cb.y)||1e-6;
  return Math.acos(Math.max(-1,Math.min(1,dot/m)))*180/Math.PI;
}
function vis(lm,i){return lm[i] && (lm[i].visibility===undefined||lm[i].visibility>0.4);}
// Landmark indices: 23/24 hip, 25/26 knee, 27/28 ankle, 11/12 shoulder
function pickSide(lm){
  const r=(lm[24]?.visibility||0)+(lm[26]?.visibility||0)+(lm[28]?.visibility||0);
  const l=(lm[23]?.visibility||0)+(lm[25]?.visibility||0)+(lm[27]?.visibility||0);
  return r>=l?{hip:24,knee:26,ankle:28,sho:12}:{hip:23,knee:25,ankle:27,sho:11};
}

/* ---------------- Main loop ---------------- */
const cv=$('cv'), ctx=cv.getContext('2d');
function loop(){
  if(!running)return;
  const v=$('vid');
  if(v.readyState>=2){
    if(cv.width!==v.videoWidth){cv.width=v.videoWidth;cv.height=v.videoHeight;}
    const now=performance.now();
    if(v.currentTime!==lastVideoTime){
      lastVideoTime=v.currentTime;
      const res=landmarker.detectForVideo(v,now);
      ctx.clearRect(0,0,cv.width,cv.height);
      if(res.landmarks&&res.landmarks.length){
        process(res.landmarks[0]);
      }else{
        coach(false,'Step back so your whole leg is in frame.');
      }
    }
  }
  rafId=requestAnimationFrame(loop);
}

function process(lm){
  const s=pickSide(lm);
  const okJoints=vis(lm,s.hip)&&vis(lm,s.knee)&&vis(lm,s.ankle);
  drawSkeleton(lm,s,okJoints);
  if(!okJoints){coach(false,'Make sure your hip, knee and ankle are all visible.');return;}

  const kneeAngle=angle(lm[s.hip],lm[s.knee],lm[s.ankle]); // 180 = straight
  const t=current.trackType;

  if(t==='flexion'){
    const bend=180-kneeAngle;                    // 0 straight … higher = more bend
    handleMetric(bend,'bend');
  }else if(t==='extension'){
    handleMetric(kneeAngle,'straight');          // closer to 180 = straighter
  }else if(t==='slr'){
    // hip flexion = angle at hip between shoulder line and knee; approximate via thigh vs torso
    const hipAngle=angle(lm[s.sho],lm[s.hip],lm[s.knee]); // 180 = leg in line with torso (flat)
    const lift=180-hipAngle;                     // how far leg raised off the bed
    const kneeStraight=kneeAngle>155;
    handleSLR(lift,kneeStraight);
  }
}

/* smoothing */
function smooth(v){ smoothAngle = smoothAngle==null? v : smoothAngle*0.6+v*0.4; return smoothAngle; }

function handleMetric(raw,kind){
  const val=Math.round(smooth(raw));
  $('angleVal').textContent=val;
  if(val>peak){peak=val;$('peakVal').textContent=peak;}
  updateNeedle(kind==='bend'?val: (kind==='straight'?val:val));
  const inRange = val>=current.goal;

  if(current.holdSec){ // hold-based (extension/quad set/stance)
    if(inRange){
      if(!holdStart)holdStart=performance.now();
      const held=(performance.now()-holdStart)/1000;
      coach(true,`Hold steady — <b>${held.toFixed(1)}s</b> / ${current.holdSec}s`);
      if(held>=current.holdSec){ countRep(); holdStart=0; }
    }else{
      holdStart=0;
      coach(false, kind==='straight'
        ? 'Push the knee straighter to reach the green zone.'
        : 'A little more — get into the green band.');
    }
  }else{ // rep-based flexion
    if(inRange && repPhase==='ready'){
      coach(true,`Great depth — now return to ${kind==='bend'?'straight':'start'}.`);
      repPhase='returning';
    }else if(repPhase==='returning' && val < current.goal*0.35){
      countRep(); repPhase='ready';
    }else if(repPhase==='ready'){
      coach(false, kind==='bend'?'Slide the heel in to reach the green zone.':'Move into the green zone.');
    }
  }
}

function handleSLR(lift,kneeStraight){
  const val=Math.round(smooth(lift));
  $('angleVal').textContent=val;
  if(val>peak){peak=val;$('peakVal').textContent=peak;}
  updateNeedle(val);
  if(!kneeStraight){ coach(false,'Keep the knee <b>locked straight</b> as you lift.'); return; }
  const inRange=val>=current.goal;
  if(inRange && repPhase==='ready'){ coach(true,'Good height — lower slowly.'); repPhase='returning'; }
  else if(repPhase==='returning' && val<8){ countRep(); repPhase='ready'; }
  else if(repPhase==='ready'){ coach(true,'Knee locked — lift the leg into the green zone.'); }
}

function countRep(){
  reps++; $('repVal').textContent=reps;
  navigator.vibrate&&navigator.vibrate(30);
  if(current.reps && reps>=current.reps && !completed){ finish(); }
}

/* ---------------- Skeleton overlay ---------------- */
const BONES=[[11,12],[11,23],[12,24],[23,24],[23,25],[25,27],[27,31],[24,26],[26,28],[28,32]];
function drawSkeleton(lm,s,ok){
  const W=cv.width,H=cv.height;
  ctx.lineWidth=Math.max(3,W*0.006);
  ctx.strokeStyle=ok?'rgba(232,100,47,.55)':'rgba(255,255,255,.32)';
  BONES.forEach(([a,b])=>{
    if(!lm[a]||!lm[b])return;
    ctx.beginPath();ctx.moveTo(lm[a].x*W,lm[a].y*H);ctx.lineTo(lm[b].x*W,lm[b].y*H);ctx.stroke();
  });
  // highlight active joint
  [s.hip,s.knee,s.ankle].forEach((i,idx)=>{
    if(!lm[i])return;
    ctx.beginPath();
    ctx.arc(lm[i].x*W,lm[i].y*H, idx===1?W*0.018:W*0.012,0,7);
    ctx.fillStyle= idx===1 ? (ok?'#e8642f':'#fff') : 'rgba(255,255,255,.85)';
    ctx.fill();
  });
}

  $('done').classList.add('show');
}
